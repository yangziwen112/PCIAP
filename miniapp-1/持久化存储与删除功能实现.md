# 持久化存储与删除功能完整实现

## 🎯 功能清单

- ✅ **收藏持久化** - 收藏数据永久保存到数据库
- ✅ **历史持久化** - 浏览历史自动保存到数据库
- ✅ **单个删除** - 用户可删除单条收藏或历史
- ✅ **批量删除** - 用户可清空所有收藏或历史
- ✅ **同步恢复** - 退出登录后重新登录数据自动恢复

### 1. 删除单条收藏

```javascript
/**
 * 删除单条收藏
 * @param {string} userId - 用户ID
 * @param {string} contentId - 内容ID
 * @returns {Promise<Object>} 操作结果
 */
async function removeFavorite({ userId, contentId }) {
  if (!contentId) return { error: 'contentId_required' }
  if (!userId) return { error: 'userId_required' }
  
  try {
    // 1. 从详细收藏表删除
    const favsColl = db.collection('favorites')
    await favsColl.where({ userId, contentId }).remove()
    
    // 2. 更新用户资料表
    const profileColl = db.collection('userProfiles')
    const profileRes = await profileColl.where({ userId }).get()
    const existing = (profileRes.data || [])[0] || {}
    
    let contentIds = existing.favorites?.contentIds || []
    contentIds = contentIds.filter(id => id !== contentId)
    
    const updateData = {
      userId,
      subscriptions: existing.subscriptions || { tagIds: [], sourceIds: [] },
      favorites: {
        contentIds,
        count: contentIds.length,
        updatedAt: Date.now()
      },
      history: existing.history || { recentIds: [], totalCount: 0 }
    }
    
    if (profileRes.data && profileRes.data.length) {
      await profileColl.where({ userId }).update({ data: updateData })
    } else {
      await profileColl.add({ data: updateData })
    }
    
    return { ok: true, count: contentIds.length }
  } catch (error) {
    console.error('删除收藏失败:', error)
    return { error: '删除失败', message: error.message }
  }
}
```

### 2. 删除浏览历史

```javascript
/**
 * 删除单条浏览历史
 * @param {string} userId - 用户ID
 * @param {string} contentId - 内容ID
 * @returns {Promise<Object>} 操作结果
 */
async function removeHistory({ userId, contentId }) {
  if (!contentId) return { error: 'contentId_required' }
  if (!userId) return { error: 'userId_required' }
  
  try {
    // 1. 从详细历史表删除
    const historyColl = db.collection('history')
    await historyColl.where({ userId, contentId }).remove()
    
    // 2. 更新用户资料表
    const profileColl = db.collection('userProfiles')
    const profileRes = await profileColl.where({ userId }).get()
    const existing = (profileRes.data || [])[0] || {}
    
    let recentIds = existing.history?.recentIds || []
    recentIds = recentIds.filter(id => id !== contentId)
    
    const updateData = {
      userId,
      subscriptions: existing.subscriptions || { tagIds: [], sourceIds: [] },
      favorites: existing.favorites || { contentIds: [], count: 0 },
      history: {
        recentIds,
        totalCount: Math.max(0, (existing.history?.totalCount || 0) - 1),
        updatedAt: Date.now()
      }
    }
    
    if (profileRes.data && profileRes.data.length) {
      await profileColl.where({ userId }).update({ data: updateData })
    } else {
      await profileColl.add({ data: updateData })
    }

  } catch (error) {

}

/**
 * 记录浏览历史（自动执行）
 * @param {string} userId - 用户ID
 * @param {string} contentId - 内容ID
 * @param {number} duration - 浏览时长（可选）
 * @returns {Promise<Object>} 操作结果
 */
async function recordHistory({ userId, contentId, duration = 0 }) {
  if (!contentId) return { error: 'contentId_required' }
  if (!userId) return { error: 'userId_required' }
  
  try {
    // 1. 添加详细历史记录
    const historyColl = db.collection('history')
    await historyColl.add({ 
      data: { 
        userId, 
        contentId, 
        ts: Date.now(),
        duration
      } 
    })
    
    // 2. 更新用户资料表
    const profileColl = db.collection('userProfiles')
    const profileRes = await profileColl.where({ userId }).get()
    const existing = (profileRes.data || [])[0] || {}
    
    let recentIds = existing.history?.recentIds || []
    
    // 移除旧记录，添加到最前面
    recentIds = recentIds.filter(id => id !== contentId)
    recentIds.unshift(contentId)
    
    // 只保留最近50条
    if (recentIds.length > 50) recentIds = recentIds.slice(0, 50)
    
    const updateData = {
      userId,
      subscriptions: existing.subscriptions || { tagIds: [], sourceIds: [] },
      favorites: existing.favorites || { contentIds: [], count: 0 },
      history: {
        recentIds,
        totalCount: (existing.history?.totalCount || 0) + 1,
        updatedAt: Date.now()
      }
    }
    
    if (profileRes.data && profileRes.data.length) {
      await profileColl.where({ userId }).update({ data: updateData })
    } else {
      await profileColl.add({ data: updateData })
    }
    
    return { ok: true }
  } catch (error) {
    console.error('记录历史失败:', error)
    return { error: '记录失败' }  // 历史记录失败不影响用户体验
  }
}
```

### 1. 详情页 - 自动记录历史

```javascript
// pages/detail/index.js
import { callApi, toast } from '../../utils/request'
import { isLoggedIn, getUserId } from '../../utils/auth'
import favoriteManager from '../../utils/favoriteManager'

Page({
  data: {
    id: '',
    detail: {},
    viewStartTime: 0
  },
  
  async onShow() {
    // 记录进入页面的时间
    this.setData({ viewStartTime: Date.now() })
    this.loadDetail()
  },
  
  async onHide() {
    // 当用户离开页面时，记录浏览历史
    if (isLoggedIn() && this.data.id) {
      const duration = Date.now() - this.data.viewStartTime
      try {
        await callApi('history/record', {
          contentId: this.data.id,
          duration: Math.max(0, duration)
        })
      } catch (error) {
        console.error('记录历史失败:', error)
        // 历史记录失败不影响用户体验
      }
    }
  },
  
  async loadDetail() {
    if (!this.data.id) return
    const res = await callApi('content/detail', { id: this.data.id })
    const detail = res.detail || {}
    
    detail.favored = favoriteManager.getFavoriteStatus(this.data.id)
    
    this.setData({ detail })
  },
  
  onToggleFavorite() {
    if (!isLoggedIn()) { 
      wx.showToast({ title: '请先登录', icon: 'none' })
      return 
    }
    
    const id = this.data.id
    callApi('user/favorite/toggle', { contentId: id }).then(() => {
      const newFavoredStatus = !this.data.detail.favored
      this.setData({ detail: { ...this.data.detail, favored: newFavoredStatus } })
      
      favoriteManager.setFavoriteStatus(id, newFavoredStatus)
      
      toast(newFavoredStatus ? '已收藏' : '已取消收藏')
    }).catch(() => toast('操作失败'))
  }
})
```

### 2. 收藏页 - 删除功能

```javascript
// pages/favorites/index.js
import { callApi, toast } from '../../utils/request'
import { isLoggedIn, getUserId } from '../../utils/auth'

Page({
  data: {
    list: [],
    page: 1,
    pageSize: 10,
    hasMore: true,
    loading: false,
    totalCount: 0,
    editMode: false  // 编辑模式
  },
  
  async onShow() {
    if (!isLoggedIn()) {
      wx.showToast({ title: '请先登录', icon: 'none' })
      setTimeout(() => {
        wx.switchTab({ url: '/pages/profile/index' })
      }, 1000)
      return
    }
    this.loadList(true)
  },
  
  onPullDownRefresh() {
    this.loadList(true).finally(() => wx.stopPullDownRefresh())
  },
  
  onReachBottom() {
    if (this.data.loading || !this.data.hasMore) return
    this.loadList(false)
  },
  
  async loadList(reset = false) {
    this.setData({ loading: true })
    const page = reset ? 1 : this.data.page + 1
    
    try {
      const res = await callApi('favorites/list', { 
        page, 
        pageSize: this.data.pageSize 
      })
      
      const list = reset ? (res.list || []) : this.data.list.concat(res.list || [])
      
      this.setData({ 
        list, 
        page, 
        hasMore: res.hasMore || false,
        totalCount: res.totalCount || 0,
        loading: false 
      })
    } catch (error) {
      console.error('加载收藏失败:', error)
      this.setData({ loading: false })
      toast('加载失败')
    }
  },
  
  // 删除单条收藏
  async onRemoveFavorite(e) {
    const id = e.currentTarget.dataset.id
    
    wx.showModal({
      title: '删除收藏',
      content: '确认删除这个收藏吗？',
      success: async (res) => {
        if (!res.confirm) return
        
        try {
          await callApi('user/favorite/remove', { contentId: id })
          
          this.setData({ 
            list: this.data.list.filter(it => it._id !== id),
            totalCount: Math.max(0, this.data.totalCount - 1)
          })
          
          toast('已删除')
        } catch (error) {
          toast('删除失败')
        }
      }
    })
  },
  
  // 清空所有收藏
  async onClearAll() {
    if (this.data.list.length === 0) {
      toast('暂无收藏')
      return
    }
    
    wx.showModal({
      title: '清空收藏',
      content: `确认删除全部 ${this.data.totalCount} 个收藏吗？此操作不可恢复`,
      confirmColor: '#ff6b6b',
      success: async (res) => {
        if (!res.confirm) return
        
        wx.showLoading({ title: '清空中...' })
        try {
          await callApi('user/favorite/clearAll', {})
          
          this.setData({ 
            list: [],
            totalCount: 0,
            editMode: false
          })
          
          wx.hideLoading()
          toast('已清空全部收藏')
        } catch (error) {
          wx.hideLoading()
          toast('清空失败')
        }
      }
    })
  },
  
  // 切换编辑模式
  onToggleEditMode() {
    this.setData({ editMode: !this.data.editMode })
  }
})
```

## 🔄 数据流程

### 收藏流程
```
1. 用户在详情页点击收藏
   ↓
2. toggleFavorite() 调用 API
   ↓
3. 后端更新 favorites 表和 userProfiles 表
   ↓
4. 用户在收藏页看到新增的收藏
   ↓
5. 用户点击编辑模式
   ↓
6. 点击删除按钮或清空全部
   ↓
7. removeFavorite() 或 clearAllFavorites() 调用 API
   ↓
8. 后端删除对应记录
   ↓
9. 收藏页实时更新显示
```

### 历史流程
```
1. 用户进入详情页
   ↓
2. onShow() 记录进入时间
   ↓
3. onHide() 时调用 recordHistory() 记录历史
   ↓
4. 后端自动保存到 history 表和 userProfiles 表
   ↓
5. 用户在历史页看到浏览记录
   ↓
6. 用户可以删除单条或清空全部
   ↓
7. removeHistory() 或 clearAllHistory() 调用 API
   ↓
8. 后端删除对应记录
   ↓
9. 历史页实时更新显示
```
