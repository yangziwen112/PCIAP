### 第1步：初始化数据库集合

在小程序首页或管理页面，添加一个新的初始化按钮：

```javascript

async function initUserDataSystem() {
  wx.showLoading({ title: '初始化用户数据系统...' })
  try {

    wx.hideLoading()
    if (res.ok) {
      wx.showToast({ title: '初始化成功！', icon: 'success' })

}
```

### 第4步：更新前端收藏页面

修改 `pages/favorites/index.js`：

```javascript
import { callApi, toast } from '../../utils/request'
import { isLoggedIn, getUserId } from '../../utils/auth'

Page({
  data: {
    list: [],
    page: 1,
    pageSize: 10,
    hasMore: true,
    loading: false,
    totalCount: 0
  },
  
  async onShow() {
    if (!isLoggedIn()) {
      wx.showToast({ title: '请先登录', icon: 'none' })
      setTimeout(() => {
        wx.switchTab({ url: '/pages/profile/index' })
      }, 1000)
      return
    }
    this.loadList(true)
  },
  
  onPullDownRefresh() {
    this.loadList(true).finally(() => wx.stopPullDownRefresh())
  },
  
  onReachBottom() {
    if (this.data.loading || !this.data.hasMore) return
    this.loadList(false)
  },
  
  async loadList(reset = false) {
    this.setData({ loading: true })
    const page = reset ? 1 : this.data.page + 1
    
    try {
      const res = await callApi('favorites/list', { 
        page, 
        pageSize: this.data.pageSize 
      })
      
      const list = reset ? (res.list || []) : this.data.list.concat(res.list || [])
      
      this.setData({ 
        list, 
        page, 
        hasMore: res.hasMore || false,
        totalCount: res.totalCount || 0,
        loading: false 
      })
    } catch (error) {
      console.error('加载收藏失败:', error)
      this.setData({ loading: false })
      toast('加载失败')
    }
  },
  
  async onUnfavorite(e) {
    const id = e.currentTarget.dataset.id
    try {
      await callApi('user/favorite/toggle', { contentId: id })
      this.setData({ 
        list: this.data.list.filter(it => it._id !== id),
        totalCount: Math.max(0, this.data.totalCount - 1)
      })
      toast('已取消收藏')
    } catch (error) {
      toast('操作失败')
    }
  }
})
```

### Q3: 如何处理数据同步问题？

如果 `userProfiles` 和 `favorites`/`history` 不同步：

```javascript
// 修复单个用户数据
async function fixUserDataSync(userId) {
  try {
    // 1. 从详细表重新生成聚合数据
    const favs = await db.collection('favorites').where({ userId }).get()
    const history = await db.collection('history').where({ userId }).get()
    const subs = await db.collection('subscriptions').where({ userId }).get()
    
    // 2. 更新到 userProfiles
    const profileData = {
      userId,
      subscriptions: (subs.data || [])[0] || { tagIds: [], sourceIds: [] },
      favorites: {
        contentIds: (favs.data || []).map(f => f.contentId),
        count: (favs.data || []).length
      },
      history: {
        recentIds: (history.data || []).map(h => h.contentId).slice(0, 50),
        totalCount: (history.data || []).length
      }
    }
    
    // 3. 保存
    await db.collection('userProfiles').where({ userId }).update({ data: profileData })
  } catch (error) {
    console.error('修复失败:', error)
  }
}
```
