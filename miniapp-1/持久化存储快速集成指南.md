// 删除单条收藏
async function removeFavorite({ userId, contentId }) {
  if (!contentId) return { error: 'contentId_required' }
  if (!userId) return { error: 'userId_required' }
  
  try {
    const favsColl = db.collection('favorites')
    await favsColl.where({ userId, contentId }).remove()
    
    const profileColl = db.collection('userProfiles')
    const profileRes = await profileColl.where({ userId }).get()
    const existing = (profileRes.data || [])[0] || {}
    
    let contentIds = existing.favorites?.contentIds || []
    contentIds = contentIds.filter(id => id !== contentId)
    
    const updateData = {
      userId,
      subscriptions: existing.subscriptions || { tagIds: [], sourceIds: [] },
      favorites: { contentIds, count: contentIds.length, updatedAt: Date.now() },
      history: existing.history || { recentIds: [], totalCount: 0 }
    }
    
    if (profileRes.data && profileRes.data.length) {
      await profileColl.where({ userId }).update({ data: updateData })
    } else {
      await profileColl.add({ data: updateData })
    }
    
    return { ok: true, count: contentIds.length }
  } catch (error) {
    console.error('删除收藏失败:', error)
    return { error: '删除失败' }
  }
}

// 删除单条历史
async function removeHistory({ userId, contentId }) {
  if (!contentId) return { error: 'contentId_required' }
  if (!userId) return { error: 'userId_required' }
  
  try {
    const historyColl = db.collection('history')
    await historyColl.where({ userId, contentId }).remove()
    
    const profileColl = db.collection('userProfiles')
    const profileRes = await profileColl.where({ userId }).get()
    const existing = (profileRes.data || [])[0] || {}

    recentIds = recentIds.filter(id => id !== contentId)
    
    const updateData = {
      userId,
      subscriptions: existing.subscriptions || { tagIds: [], sourceIds: [] },
      favorites: existing.favorites || { contentIds: [], count: 0 },
      history: { recentIds, totalCount: Math.max(0, (existing.history?.totalCount || 0) - 1), updatedAt: Date.now() }
    }
    
    if (profileRes.data && profileRes.data.length) {
      await profileColl.where({ userId }).update({ data: updateData })
    } else {

// 记录浏览历史（自动执行）
async function recordHistory({ userId, contentId, duration = 0 }) {
  if (!contentId || !userId) return { error: '参数错误' }
  
  try {
    const historyColl = db.collection('history')
    await historyColl.add({ data: { userId, contentId, ts: Date.now(), duration } })
    
    const profileColl = db.collection('userProfiles')
    const profileRes = await profileColl.where({ userId }).get()
    const existing = (profileRes.data || [])[0] || {}
    
    let recentIds = existing.history?.recentIds || []
    recentIds = recentIds.filter(id => id !== contentId)
    recentIds.unshift(contentId)
    if (recentIds.length > 50) recentIds = recentIds.slice(0, 50)
    
    const updateData = {
      userId,
      subscriptions: existing.subscriptions || { tagIds: [], sourceIds: [] },
      favorites: existing.favorites || { contentIds: [], count: 0 },
      history: { recentIds, totalCount: (existing.history?.totalCount || 0) + 1, updatedAt: Date.now() }
    }
    
    if (profileRes.data && profileRes.data.length) {
      await profileColl.where({ userId }).update({ data: updateData })
    } else {
      await profileColl.add({ data: updateData })
    }
    
    return { ok: true }
  } catch (error) {
    return { error: '记录失败' }
  }
}

import { callApi, toast } from '../../utils/request'
import { isLoggedIn, getUserId } from '../../utils/auth'
import favoriteManager from '../../utils/favoriteManager'

Page({
  data: {
    id: '',
    detail: {},
    viewStartTime: 0
  },
  
  async onShow() {
    this.setData({ viewStartTime: Date.now() })
    this.loadDetail()
  },
  
  // 自动记录浏览历史
  async onHide() {
    if (isLoggedIn() && this.data.id) {
      const duration = Date.now() - this.data.viewStartTime
      callApi('history/record', { contentId: this.data.id, duration }).catch(() => {})
    }
  },
  
  async loadDetail() {
    if (!this.data.id) return
    const res = await callApi('content/detail', { id: this.data.id })
    const detail = res.detail || {}
    detail.favored = favoriteManager.getFavoriteStatus(this.data.id)
    this.setData({ detail })
  },
  
  onToggleFavorite() {
    if (!isLoggedIn()) { wx.showToast({ title: '请先登录', icon: 'none' }); return }
    const id = this.data.id
    callApi('user/favorite/toggle', { contentId: id }).then(() => {
      const newFavoredStatus = !this.data.detail.favored
      this.setData({ detail: { ...this.data.detail, favored: newFavoredStatus } })
      favoriteManager.setFavoriteStatus(id, newFavoredStatus)
      toast(newFavoredStatus ? '已收藏' : '已取消收藏')
    }).catch(() => toast('操作失败'))
  },
  
  onOpenSource() {
    const url = this.data.detail.sourceUrl
    if (!url) return
    if (url.startsWith('http')) {
      wx.setClipboardData({ data: url, success: () => wx.showToast({ title: '链接已复制', icon: 'none' }) })
    }
  }
})

### 第 4 步：更新收藏页（10 分钟）

**打开** `pages/favorites/index.js`

**替换为**：

```javascript
import { callApi, toast } from '../../utils/request'
import { isLoggedIn } from '../../utils/auth'

Page({
  data: {
    list: [],
    page: 1,
    pageSize: 10,
    hasMore: true,
    loading: false,
    totalCount: 0,
    editMode: false
  },
  
  async onShow() {
    if (!isLoggedIn()) {
      wx.showToast({ title: '请先登录', icon: 'none' })
      setTimeout(() => wx.switchTab({ url: '/pages/profile/index' }), 1000)
      return
    }
    this.loadList(true)
  },
  
  onPullDownRefresh() {
    this.loadList(true).finally(() => wx.stopPullDownRefresh())
  },
  
  onReachBottom() {
    if (this.data.loading || !this.data.hasMore) return
    this.loadList(false)
  },
  
  async loadList(reset = false) {
    this.setData({ loading: true })
    const page = reset ? 1 : this.data.page + 1
    try {
      const res = await callApi('favorites/list', { page, pageSize: this.data.pageSize })
      const list = reset ? (res.list || []) : this.data.list.concat(res.list || [])
      this.setData({ list, page, hasMore: res.hasMore || false, totalCount: res.totalCount || 0, loading: false })
    } catch (error) {
      this.setData({ loading: false })
      toast('加载失败')
    }
  },
  
  async onRemoveFavorite(e) {
    const id = e.currentTarget.dataset.id
    wx.showModal({
      title: '删除收藏',
      content: '确认删除吗？',
      success: async (res) => {
        if (!res.confirm) return
        try {
          await callApi('user/favorite/remove', { contentId: id })
          this.setData({ 
            list: this.data.list.filter(it => it._id !== id),
            totalCount: Math.max(0, this.data.totalCount - 1)
          })
          toast('已删除')
        } catch (error) {
          toast('删除失败')
        }
      }
    })
  },
  
  async onClearAll() {
    wx.showModal({
      title: '清空收藏',
      content: `确认删除全部 ${this.data.totalCount} 个收藏吗？`,
      confirmColor: '#ff6b6b',
      success: async (res) => {
        if (!res.confirm) return
        wx.showLoading({ title: '清空中...' })
        try {
          await callApi('user/favorite/clearAll', {})
          this.setData({ list: [], totalCount: 0, editMode: false })
          wx.hideLoading()
          toast('已清空')
        } catch (error) {
          wx.hideLoading()
          toast('清空失败')
        }
      }
    })
  },
  
  onToggleEditMode() {
    this.setData({ editMode: !this.data.editMode })
  }
})
```
